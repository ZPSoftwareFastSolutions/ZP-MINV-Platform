Option Explicit

' Doble clic en un producto: formulario de registro listo para su reposición (ENTRADA).
Private Sub Worksheet_BeforeDoubleClick(ByVal Target As Range, Cancel As Boolean)
    Dim sku As String
    sku = SkuDeFila(Me, Target.Row)
    If Len(sku) = 0 Then Exit Sub
    Cancel = True
    AbrirFormularioReposicion sku
End Sub
