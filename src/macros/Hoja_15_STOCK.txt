Option Explicit

' Doble clic en un producto: abre su consulta (kardex).
Private Sub Worksheet_BeforeDoubleClick(ByVal Target As Range, Cancel As Boolean)
    Dim sku As String
    sku = SkuDeFila(Me, Target.Row)
    If Len(sku) = 0 Then Exit Sub
    Cancel = True
    AbrirKardex sku
End Sub
